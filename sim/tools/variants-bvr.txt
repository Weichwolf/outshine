# FlightBox — intercept-pilot VARIANTS for tools/fb_tournament.py.
#
# One variant per line: `<name> pilot_key=value ...` — the keys are systems/FBPilotTuning's table, i.e.
# exactly the `set pilot_*` lines the tournament writes into the mission files it generates
# (doc/missions/combat.md). A bare name flies the airframe's own numbers (modules/f16/FBF16Pilot).
#
# ONE LEVER PER VARIANT, on purpose: a ranking is only readable as an explanation if the thing that
# changed is a single decision. Each line below is a doctrine somebody argues for in a debrief.

baseline                              # FBF16Pilot as it stands: designate at 16 nm, shoot at Rtr,
                                      # crank 45 deg, 1.0 s to answer a warning
longshot   pilot_shot_rtr=1.6         # MAXIMUM RANGE: pull the trigger at 1.6x Rtr — a shot the
                                      # target can outrun, taken to put him on the defensive first
deepshot   pilot_shot_rtr=0.7         # THE OPPOSITE: hold the trigger until deep inside Rtr, where
                                      # the round arrives whatever he does — if he lets you get there
earlylock  pilot_lock_nm=26           # spend the lock early: a long-settled track before the shot
                                      # range arrives, at the price of warning him far sooner
latelock   pilot_lock_nm=9            # spend it as late as possible: minimum warning, minimum time
slowhand   pilot_react_s=4.0          # a pilot who takes four seconds to answer a launch warning
