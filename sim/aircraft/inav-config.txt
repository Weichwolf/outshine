# iNav SITL SHARED BASE for the FlightBox X-Plane FDM bridge (CLI, over TCP 5760).
# This file carries ONLY what every aircraft shares: sensors, receiver/protocol, arming safety,
# failsafe, and the flight-mode channel map. Everything airframe-specific — servo/motor MIX, control
# surfaces, flap/gear policy, and the NAV speed/climb/loiter tuning — lives in the model plugin
# (models/<name>/inav.diff), applied on top by make-eeprom.sh. A jet and a glider share this base but
# nothing of their control config. Apply order: inav-config.txt, then models/<ac>/inav.diff, then save.
#
#   SITL.elf --path=eeprom.bin --sim=xp --simip=127.0.0.1 --simport=49000 --chanmap=<from profile.env>

# --- platform + receiver ---
set platform_type = AIRPLANE
set receiver_type = MSP

# --- sensors: truth-attitude flies stable (attitude given, no AHRS lag); the mag need only be PRESENT
#     so navIsHeadingUsable=true (SENSOR_MAG path) -> native NAV works, vanilla iNav, no patch, no --useimu.
set mag_hardware = FAKE
feature GPS
set gps_provider = FAKE

# --- arming safety + failsafe (shared policy) ---
set nav_extra_arming_safety = ALLOW_BYPASS   ; arm with YAW stick HIGH to bypass NAVIGATION_UNSAFE
set failsafe_procedure = RTH
set failsafe_delay = 5

# --- flight-mode channel map: ARM=AUX1(pid0), ANGLE=AUX2(pid1), NAV RTH=AUX3(pid10), NAV WP=AUX4(pid28).
#     aux uses iNav permanentId, NOT the box enum (NAV RTH=10, NAV WP=28, fc_msp_box.c). Shared by all. ---
aux 0 0 0 1700 2100
aux 1 1 1 1700 2100
aux 2 10 2 1700 2100
aux 3 28 3 1700 2100

# --- test convenience ---
set small_angle = 180

# --- headless-sim speed: a slower gyro loop (500 Hz vs 1 kHz) halves the SITL scheduler load, so the
#     clock shim can run FB_TIME_SCALE higher before iNav flags SYSTEM_OVERLOADED (arming-disable).
#     500 Hz is ample for a fixed-wing model; this changes sim throughput, not flight behaviour. ---
set looptime = 2000

save
