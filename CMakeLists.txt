cmake_minimum_required(VERSION 3.28)
project(outshine LANGUAGES CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
foreach(configuration RELEASE RELWITHDEBINFO MINSIZEREL)
  string(REPLACE "-DNDEBUG" "" CMAKE_CXX_FLAGS_${configuration} "${CMAKE_CXX_FLAGS_${configuration}}")
  if(APPLE)
    string(REPLACE "-DNDEBUG" "" CMAKE_OBJCXX_FLAGS_${configuration} "${CMAKE_OBJCXX_FLAGS_${configuration}}")
  endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(CURL REQUIRED)
find_package(osqp CONFIG REQUIRED)
find_package(highs CONFIG REQUIRED)
find_package(qjs CONFIG REQUIRED)
foreach(package sdl3 sdl3-image sdl3-ttf libtiff-4 libwebp geos sqlite3 libzstd nlohmann_json)
  string(MAKE_C_IDENTIFIER "${package}" identifier)
  pkg_check_modules(${identifier} REQUIRED IMPORTED_TARGET ${package})
endforeach()
pkg_check_modules(spirv_cross QUIET IMPORTED_TARGET spirv-cross-c-shared)
if(NOT spirv_cross_FOUND)
  pkg_check_modules(spirv_cross REQUIRED IMPORTED_TARGET spirv-cross-c)
  foreach(component msl hlsl glsl reflect cpp core)
    find_library(spirv_cross_${component} NAMES spirv-cross-${component} REQUIRED)
    list(APPEND SPIRV_CROSS_LIBRARIES "${spirv_cross_${component}}")
  endforeach()
endif()
if(NOT APPLE)
  find_package(Vulkan REQUIRED)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${Vulkan_INCLUDE_DIR}/vulkan/vulkan_core.h")
  try_compile(OUTSHINE_VULKAN_PRESENT_HEADERS
    SOURCE_FROM_CONTENT VulkanPresentHeaders.cpp [=[
#include <vulkan/vulkan_core.h>

static_assert(VK_KHR_internally_synchronized_queues != 0);
static_assert(VK_DEVICE_QUEUE_CREATE_INTERNALLY_SYNCHRONIZED_BIT_KHR != 0);

int main() {
  const VkPhysicalDeviceInternallySynchronizedQueuesFeaturesKHR features{
      .sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_INTERNALLY_SYNCHRONIZED_QUEUES_FEATURES_KHR,
      .pNext = nullptr,
      .internallySynchronizedQueues = VK_FALSE};
  return static_cast<int>(features.internallySynchronizedQueues);
}
]=]
    LINK_LIBRARIES Vulkan::Headers
    NO_CACHE)
  if(NOT OUTSHINE_VULKAN_PRESENT_HEADERS)
    message(FATAL_ERROR
      "Vulkan headers must support VK_KHR_internally_synchronized_queues for single-queue presentation. Install current Vulkan development headers or set -DVulkan_INCLUDE_DIR=<SDK>/include.")
  endif()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
find_program(CCACHE_EXECUTABLE ccache)
if(CCACHE_EXECUTABLE)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_EXECUTABLE}")
  set(CMAKE_OBJCXX_COMPILER_LAUNCHER "${CCACHE_EXECUTABLE}")
endif()
set(OUTSHINE_SANITIZER "" CACHE STRING "Native instrumentation: address or thread")
set_property(CACHE OUTSHINE_SANITIZER PROPERTY STRINGS "" address thread)
option(OUTSHINE_HARDENED "Enable checked standard-library contracts" OFF)
set(OUTSHINE_TEST_TIMEOUT 120 CACHE STRING "CTest timeout in seconds")
option(OUTSHINE_CLANG_TIDY "Run clang-tidy through CMake's native compilation rule" OFF)
if(OUTSHINE_CLANG_TIDY)
  find_program(CLANG_TIDY_EXECUTABLE NAMES clang-tidy clang-tidy-23 clang-tidy-22
    HINTS /opt/homebrew/opt/llvm/bin REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_EXECUTABLE};--quiet")
  set(CMAKE_OBJCXX_CLANG_TIDY "${CLANG_TIDY_EXECUTABLE};--quiet")
endif()
include(cmake/Compiler.cmake)
include(cmake/Components.cmake)
include(cmake/Generated.cmake)
include(cmake/Shaders.cmake)
add_library(outshine_core STATIC)
foreach(component IN LISTS OUTSHINE_COMPONENTS)
  if(NOT component STREQUAL "outshine_diagnostics")
    target_sources(outshine_core PRIVATE $<TARGET_OBJECTS:${component}_objects>)
  endif()
endforeach()
target_link_libraries(outshine_core INTERFACE outshine_client)
set_target_properties(outshine_core PROPERTIES OUTPUT_NAME outshine)
set_target_properties(outshine_core PROPERTIES LINKER_LANGUAGE CXX)
add_executable(outshine src/client/Main.cpp)
target_link_libraries(outshine PRIVATE outshine_core)
add_dependencies(outshine outshine_shaders)
include(CTest)
if(BUILD_TESTING)
  include(cmake/Tests.cmake)
endif()
